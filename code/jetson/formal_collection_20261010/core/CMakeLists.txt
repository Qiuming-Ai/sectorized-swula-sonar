cmake_minimum_required(VERSION 3.18)
project(SonarFormalCollection20261010 LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 87)
endif()
find_package(CUDAToolkit REQUIRED)
get_filename_component(AUDIT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(V2_CORE "${AUDIT_ROOT}/implementation_v2/core")
set(V3_CORE "${AUDIT_ROOT}/implementation_v3/core")
# Compile frozen engines without changing their source files.
add_library(formal_v2 STATIC "${V2_CORE}/engine.cu")
target_include_directories(formal_v2 PUBLIC "${V2_CORE}")
target_link_libraries(formal_v2 PUBLIC CUDA::cudart CUDA::cufft)
add_library(formal_v3 STATIC "${V3_CORE}/engine_v3.cu")
target_include_directories(formal_v3 PUBLIC "${V2_CORE}" "${V3_CORE}")
target_link_libraries(formal_v3 PUBLIC CUDA::cudart CUDA::cufft)
add_executable(formal_runner main.cpp)
target_link_libraries(formal_runner PRIVATE formal_v2 formal_v3)
target_compile_options(formal_runner PRIVATE -Wall -Wextra -Wpedantic)
